darboux attendus : x (cofacteur 1 - y), y (cofacteur x - 1)

sage -c '__name__ = "lib"; load("extactique.sage"); P, Q = x*(1-y), y*(x-1); [print("n = %d :" % n, candidats_darboux(P, Q, n)[1]) for n in [2, 3]]'

n = 2 : [(y, x - 1), (x, -y + 1)]
